class DbHelper {
  static final DbHelper _instance = DbHelper._internal();
  factory DbHelper() => _instance;
  DbHelper._internal();

  static Database? _database;

  Future<Database> get database async {
    if (_database != null) return _database!;
    _database = await _initDb();
    return _database!;
  }

  Future<Database> _initDb() async {
    final dbPath = await getDatabasesPath();
    final path = join(dbPath, 'alacena.db');

    return await openDatabase(
      path,
      version: 1,
      onConfigure: (db) async {
        await db.execute('PRAGMA foreign_keys = ON');
      },
      onCreate: _onCreate,
    );
  }

  Future<void> _onCreate(Database db, int version) async {
    // 1. Tablas
    await db.execute('''
      CREATE TABLE categoria (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        nombre TEXT NOT NULL UNIQUE CHECK (length(trim(nombre)) > 0),
        icono TEXT
      )
    ''');

    await db.execute('''
      CREATE TABLE tienda (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        nombre TEXT NOT NULL UNIQUE CHECK (length(trim(nombre)) > 0),
        direccion TEXT
      )
    ''');

    await db.execute('''
      CREATE TABLE producto (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        nombre TEXT NOT NULL CHECK (length(trim(nombre)) > 0),
        categoria_id INTEGER NOT NULL,
        imagen_path TEXT,
        unidad TEXT NOT NULL DEFAULT 'pza',
        stock_minimo INTEGER NOT NULL DEFAULT 1 CHECK (stock_minimo >= 0),
        ultima_tienda_id INTEGER,
        FOREIGN KEY (categoria_id) REFERENCES categoria(id) ON DELETE RESTRICT,
        FOREIGN KEY (ultima_tienda_id) REFERENCES tienda(id) ON DELETE SET NULL
      )
    ''');

    await db.execute('''
      CREATE TABLE lote (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        producto_id INTEGER NOT NULL,
        cantidad INTEGER NOT NULL CHECK (cantidad >= 0),
        fecha_caducidad TEXT,
        alarma_activa INTEGER NOT NULL DEFAULT 1 CHECK (alarma_activa IN (0,1)),
        fecha_ingreso TEXT NOT NULL,
        FOREIGN KEY (producto_id) REFERENCES producto(id) ON DELETE CASCADE
      )
    ''');

    await db.execute('''
      CREATE TABLE historial_compras (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        producto_id INTEGER NOT NULL,
        tienda_id INTEGER NOT NULL,
        cantidad INTEGER NOT NULL CHECK (cantidad > 0),
        precio REAL CHECK (precio IS NULL OR precio >= 0),
        fecha_compra TEXT NOT NULL,
        FOREIGN KEY (producto_id) REFERENCES producto(id) ON DELETE CASCADE,
        FOREIGN KEY (tienda_id) REFERENCES tienda(id) ON DELETE RESTRICT
      )
    ''');

    await db.execute('''
      CREATE TABLE carrito_item (
        id INTEGER PRIMARY KEY AUTOINCREMENT,
        producto_id INTEGER NOT NULL UNIQUE,
        cantidad INTEGER NOT NULL DEFAULT 1 CHECK (cantidad > 0),
        FOREIGN KEY (producto_id) REFERENCES producto(id) ON DELETE CASCADE
      )
    ''');

    // 2. Ãndices
    await db.execute('CREATE INDEX idx_lote_producto ON lote(producto_id)');
    await db.execute('CREATE INDEX idx_lote_caducidad ON lote(fecha_caducidad)');
    await db.execute('CREATE INDEX idx_historial_producto ON historial_compras(producto_id)');

    // 3. Semilla (Datos iniciales)
    await _insertarSemilla(db);
  }

  Future<void> _insertarSemilla(Database db) async {
    final categorias = ['LÃ¡cteos', 'Carnes', 'Frutas y verduras', 'Despensa', 'Bebidas', 'Limpieza'];
    for (var cat in categorias) {
      await db.insert('categoria', {'nombre': cat});
    }

    final tiendas = [
      {'nombre': 'Walmart', 'direccion': 'Sucursal Centro'},
      {'nombre': 'Soriana', 'direccion': 'Sucursal Norte'},
    ];
    for (var t in tiendas) {
      await db.insert('tienda', t);
    }
  }
}

// ARCHIVO: calendario_repository_impl.dart
// ==========================================

import '../../core/date_utils.dart';
import '../models/producto_resumen.dart';
import 'calendario_repository.dart';
import 'producto_repository.dart';

class CalendarioRepositoryImpl implements CalendarioRepository {
  final ProductoRepository _productoRepo;

  CalendarioRepositoryImpl(this._productoRepo);

  @override
  Future<Map<DateTime, List<ProductoResumen>>> getAlertasPorDia() async {
    final alertas = await _productoRepo.getAlertas();
    final Map<DateTime, List<ProductoResumen>> mapa = {};

    for (var p in alertas) {
      if (p.proximaCaducidad != null) {
        final fecha = normalizarFecha(p.proximaCaducidad!);
        if (!mapa.containsKey(fecha)) mapa[fecha] = [];
        mapa[fecha]!.add(p);
      }
    }
    return mapa;
  }

  @override
  Future<List<ProductoResumen>> getAlertasDeDia(DateTime dia) async {
    final mapa = await getAlertasPorDia();
    return mapa[normalizarFecha(dia)] ?? [];
  }
}

// ARCHIVO: carrito_repository_impl.dart
// ==========================================

import 'dart:async';
import '../database/db_helper.dart';
import '../models/carrito_item.dart';
import 'carrito_repository.dart';

class CarritoRepositoryImpl implements CarritoRepository {
  final DbHelper _dbHelper = DbHelper();
  final _controller = StreamController<int>.broadcast();

  CarritoRepositoryImpl() {
    _emitirConteo(); // Carga inicial
  }

  Future<void> _emitirConteo() async {
    final db = await _dbHelper.database;
    final res = await db.rawQuery('SELECT SUM(cantidad) as total FROM carrito_item');
    final total = (res.first['total'] as int?) ?? 0;
    _controller.add(total);
  }

  @override
  Stream<int> watchCantidad() => _controller.stream;

  @override
  Future<List<CarritoItem>> getItems() async {
    final db = await _dbHelper.database;
    final res = await db.query('carrito_item');
    return res.map((e) => CarritoItem.fromMap(e)).toList();
  }

  @override
  Future<void> agregar(int productoId, {int cantidad = 1}) async {
    final db = await _dbHelper.database;
    
    // Verificar si ya existe
    final res = await db.query('carrito_item', where: 'producto_id = ?', whereArgs: [productoId]);
    if (res.isNotEmpty) {
      final item = CarritoItem.fromMap(res.first);
      await db.update(
        'carrito_item', 
        {'cantidad': item.cantidad + cantidad}, 
        where: 'id = ?', 
        whereArgs: [item.id]
      );
    } else {
      await db.insert('carrito_item', {'producto_id': productoId, 'cantidad': cantidad});
    }
    _emitirConteo();
  }

  @override
  Future<void> quitar(int productoId) async {
    final db = await _dbHelper.database;
    await db.delete('carrito_item', where: 'producto_id = ?', whereArgs: [productoId]);
    _emitirConteo();
  }

  @override
  Future<void> vaciar() async {
    final db = await _dbHelper.database;
    await db.delete('carrito_item');
    _emitirConteo();
  }
}

import '../database/db_helper.dart';
import '../models/historial_compra.dart';
import '../models/lote.dart';
import '../models/tienda.dart';
import 'compra_repository.dart';

class CompraRepositoryImpl implements CompraRepository {
  final DbHelper _dbHelper = DbHelper();

  @override
  Future<void> registrarCompraConLote(HistorialCompra c, Lote l) async {
    final db = await _dbHelper.database;
    
    await db.transaction((txn) async {
      // 1. Insertar compra
      await txn.insert('historial_compras', c.toMap());
      
      // 2. Actualizar Ãºltima tienda del producto
      await txn.update(
        'producto', 
        {'ultima_tienda_id': c.tiendaId}, 
        where: 'id = ?', 
        whereArgs: [c.productoId]
      );
      
      // 3. Insertar lote
      await txn.insert('lote', l.toMap());
    });
  }

  @override
  Future<List<Tienda>> getTiendasDeProducto(int productoId) async {
    final db = await _dbHelper.database;
    final res = await db.rawQuery('''
      SELECT DISTINCT t.* FROM tienda t
      INNER JOIN historial_compras h ON t.id = h.tienda_id
      WHERE h.producto_id = ?
    ''', [productoId]);
    
    return res.map((e) => Tienda.fromMap(e)).toList();
  }

  @override
  Future<List<HistorialCompra>> getHistorial({int? productoId}) async {
    final db = await _dbHelper.database;
    final res = await db.query(
      'historial_compras',
      where: productoId != null ? 'producto_id = ?' : null,
      whereArgs: productoId != null ? [productoId] : null,
      orderBy: 'fecha_compra DESC',
    );
    return res.map((e) => HistorialCompra.fromMap(e)).toList();
  }
}

import '../../core/constants.dart';

import '../../core/errors.dart';

import '../models/estado_producto.dart';

class ProductoRepositoryImpl implements ProductoRepository {
  final DbHelper _dbHelper = DbHelper();

  @override
  Future<List<ProductoResumen>> getAlertas() async {
    final todos = await getTodos();
    return todos.where((p) => p.estado != EstadoProducto.normal).toList();
  }

  @override
  Future<List<ProductoResumen>> getTodos({int? categoriaId}) async {
    final db = await _dbHelper.database;
    
    String where = '';
    List<dynamic> whereArgs = [];
    if (categoriaId != null) {
      where = 'WHERE p.categoria_id = ?';
      whereArgs.add(categoriaId);
    }

    final res = await db.rawQuery('''
      SELECT p.*, c.nombre as cat_nombre, t.nombre as tienda_nombre,
             (SELECT SUM(cantidad) FROM lote WHERE producto_id = p.id AND cantidad > 0) as stock_total,
             (SELECT MIN(fecha_caducidad) FROM lote WHERE producto_id = p.id AND cantidad > 0) as prox_caducidad
      FROM producto p
      LEFT JOIN categoria c ON p.categoria_id = c.id
      LEFT JOIN tienda t ON p.ultima_tienda_id = t.id
      $where
    ''', whereArgs);

    List<ProductoResumen> lista = [];
    final hoyDate = hoy();

    for (var row in res) {
      final p = Producto.fromMap(row);
      final stockTotal = (row['stock_total'] as int?) ?? 0;
      final proxCadStr = row['prox_caducidad'] as String?;
      final proxCad = proxCadStr != null ? sqlToDate(proxCadStr) : null;

      EstadoProducto estado = EstadoProducto.normal;

      if (stockTotal == 0) {
        estado = EstadoProducto.sinExistencia;
      } else if (proxCad != null && proxCad.isBefore(hoyDate)) {
        estado = EstadoProducto.caducado;
      } else if (proxCad != null && proxCad.difference(hoyDate).inDays <= kDiasPorCaducar) {
        estado = EstadoProducto.porCaducar;
      } else if (stockTotal <= p.stockMinimo) {
        estado = EstadoProducto.porAgotarse;
      }

      lista.add(ProductoResumen(
        producto: p,
        categoriaNombre: row['cat_nombre'] as String? ?? 'Sin categorÃ­a',
        stockTotal: stockTotal,
        proximaCaducidad: proxCad,
        estado: estado,
        ultimaTiendaNombre: row['tienda_nombre'] as String?,
      ));
    }
    return lista;
  }

  @override
  Future<ProductoResumen?> getById(int id) async {
    final lista = await getTodos();
    try {
      return lista.firstWhere((p) => p.producto.id == id);
    } catch (_) {
      return null;
    }
  }

  @override
  Future<int> insertar(Producto p) async {
    final db = await _dbHelper.database;
    return await db.insert(DbTables.producto, p.toMap());
  }

  @override
  Future<void> actualizar(Producto p) async {
    final db = await _dbHelper.database;
    await db.update(DbTables.producto, p.toMap(), where: 'id = ?', whereArgs: [p.id]);
  }

  @override
  Future<void> eliminar(int id) async {
    final db = await _dbHelper.database;
    await db.delete(DbTables.producto, where: 'id = ?', whereArgs: [id]);
  }

  @override
  Future<void> consumirStock(int productoId, int cantidad) async {
    if (cantidad <= 0) return;
    final db = await _dbHelper.database;

    await db.transaction((txn) async {
      // 1. Verificar stock total
      final res = await txn.rawQuery(
          'SELECT SUM(cantidad) as total FROM lote WHERE producto_id = ? AND cantidad > 0',
          [productoId]);
      final total = (res.first['total'] as int?) ?? 0;

      if (total < cantidad) {
        throw AppException('No hay suficiente stock. Stock actual: $total');
      }

      // 2. Obtener lotes ordenados por FEFO (First Expired, First Out)
      final lotesRes = await txn.query(
        DbTables.lote,
        where: 'producto_id = ? AND cantidad > 0',
        whereArgs: [productoId],
        orderBy: 'fecha_caducidad ASC, id ASC', // FEFO
      );

      int restante = cantidad;
      for (var row in lotesRes) {
        if (restante <= 0) break;
        
        final loteId = row['id'] as int;
        final loteCant = row['cantidad'] as int;

        if (loteCant > restante) {
          await txn.update(DbTables.lote, {'cantidad': loteCant - restante}, where: 'id = ?', whereArgs: [loteId]);
          restante = 0;
        } else {
          await txn.update(DbTables.lote, {'cantidad': 0}, where: 'id = ?', whereArgs: [loteId]);
          restante -= loteCant;
        }
      }
    });
  }
}

import 'package:flutter_local_notifications/flutter_local_notifications.dart';
import 'package:timezone/data/latest_all.dart' as tz;
import 'package:timezone/timezone.dart' as tz;

class NotificationService {
  static final FlutterLocalNotificationsPlugin _plugin = FlutterLocalNotificationsPlugin();

  static Future<void> init() async {
    tz.initializeTimeZones();
    
    const androidSettings = AndroidInitializationSettings('@mipmap/ic_launcher');
    const iosSettings = DarwinInitializationSettings(
      requestAlertPermission: true,
      requestBadgePermission: true,
      requestSoundPermission: true,
    );

    const initSettings = InitializationSettings(android: androidSettings, iOS: iosSettings);
    await _plugin.initialize(initSettings);
  }

  static Future<void> programarAlarma(int id, String producto, DateTime fecha) async {
    final tzFecha = tz.TZDateTime.from(fecha, tz.local).add(const Duration(hours: 8)); // 8:00 AM
    
    // Si la fecha ya pasÃ³, no programar
    if (tzFecha.isBefore(tz.TZDateTime.now(tz.local))) return;

    await _plugin.zonedSchedule(
      id,
      'Producto por caducar',
      'El producto "$producto" caduca hoy.',
      tzFecha,
      const NotificationDetails(
        android: AndroidNotificationDetails('caducidad_channel', 'Caducidad',
            importance: Importance.max, priority: Priority.high),
        iOS: DarwinNotificationDetails(),
      ),
      androidScheduleMode: AndroidScheduleMode.exactAllowWhileIdle,
      uiLocalNotificationDateInterpretation: UILocalNotificationDateInterpretation.absoluteTime,
      payload: 'caducidad',
    );
  }

  static Future<void> cancelarTodas() async {
    await _plugin.cancelAll();
  }
}

    diaEnfocado = enfocado;
    notifyListeners();
  }

}
